# Batch LLM Gap-Fill Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Zastąpić inline LLM w `collect_rss` kolejką DB i batch jobem co 2h, z entry_price z momentu publikacji artykułu.

**Architecture:** `collect_rss` enqueueuje gap-itemy (reddit+news) do `llm_queue`. `_llm_batch_job` co 2h pobiera 8 itemów, sprawdza cache, wywołuje LLM jeśli potrzeba, tworzy claimy z `entry_price = get_historical_price(asset, item.published_at)`.

**Tech Stack:** SQLAlchemy 2.0, APScheduler, feedparser, httpx (przez istniejące moduły argus).

```python
import logging
import os
from datetime import datetime, timezone

from sqlalchemy import select, case

from argus.models import Claim, ClaimAudit, Item, LlmQueue, Source
from argus.llm.cache import cache_get, cache_put, text_hash
from argus.llm.extractor import LLM_DEFAULT_CONFIDENCE, extract_claims_llm
from argus.prices import get_historical_price, get_spot_price

logger = logging.getLogger(__name__)

_MAX_ATTEMPTS = int(os.getenv("ARGUS_LLM_MAX_ATTEMPTS", "5"))


def enqueue_for_llm(session, item, source_kind: str, text: str) -> bool:
    h = text_hash(text)
    if session.scalar(select(LlmQueue).where(LlmQueue.item_id == item.id)):
        return False
    session.add(LlmQueue(
        item_id=item.id,
        text_hash=h,
        source_kind=source_kind,
        status="pending",
        attempts=0,
        enqueued_at=datetime.now(timezone.utc),
    ))
    return True


def process_llm_batch(session, batch_size: int, assets, sector_map) -> dict:
    by_symbol = {a.symbol: a for a in assets}

    pending = list(session.scalars(
        select(LlmQueue)
        .where(LlmQueue.status == "pending", LlmQueue.attempts < _MAX_ATTEMPTS)
        .order_by(
            case((LlmQueue.source_kind == "reddit", 0), else_=1).asc(),
            LlmQueue.enqueued_at.desc(),
        )
        .limit(batch_size)
    ))

    stats = {"processed": 0, "cache_hits": 0, "llm_calls": 0, "claims": 0, "errors": 0}

    for entry in pending:
        item = session.get(Item, entry.item_id)
        if item is None:
            entry.status = "failed"
            entry.last_error = "item not found"
            session.commit()
            stats["errors"] += 1
            continue

        cached = cache_get(session, entry.text_hash)
        if cached is not None:
            triples = [(c["asset"], c["direction"], c.get("reasoning", "")) for c in cached]
            stats["cache_hits"] += 1
        else:
            try:
                llm_claims = extract_claims_llm(item.text or "", assets, sector_map)
                cache_put(session, entry.text_hash,
                          [{"asset": c.asset_symbol, "direction": c.direction, "reasoning": c.reasoning}
                           for c in llm_claims])
                triples = [(c.asset_symbol, c.direction, c.reasoning) for c in llm_claims]
                stats["llm_calls"] += 1
            except Exception as e:
                entry.attempts += 1
                entry.last_error = str(e)[:300]
                if entry.attempts >= _MAX_ATTEMPTS:
                    entry.status = "failed"
                session.commit()
                stats["errors"] += 1
                continue

        ts = item.published_at or item.fetched_at or datetime.now(timezone.utc)

        for sym, direction, reasoning in triples:
            asset = by_symbol.get(sym)
            if asset is None:
                continue
            try:
                entry_price = get_historical_price(asset, ts)
            except Exception:
                try:
                    entry_price = get_spot_price(asset)
                except Exception as e2:
                    logger.warning("skip llm claim %s: price failed: %s", sym, e2)
                    continue

            claim = Claim(
                source_id=item.source_id,
                item_id=item.id,
                asset=sym,
                direction=direction,
                confidence=LLM_DEFAULT_CONFIDENCE,
                method="llm",
                entry_price=entry_price,
                entry_ts=ts,
                status="open",
            )
            session.add(claim)
            session.flush()
            session.add(ClaimAudit(
                claim_id=claim.id,
                raw_text=item.text or "",
                parsed_claim={"asset": sym, "direction": direction, "reasoning": reasoning},
                parser_version="llm-v1",
            ))
            stats["claims"] += 1

        entry.status = "done"
        entry.processed_at = datetime.now(timezone.utc)
        session.commit()
        stats["processed"] += 1

    return stats
```

```python
import os
import calendar
import logging
from datetime import datetime, timezone
from dataclasses import asdict
import yaml
import feedparser
from sqlalchemy import select
from argus.models import Asset, Item, Claim, ClaimAudit
from argus.extract import extract_claims, PARSER_VERSION
from argus.prices import get_spot_price
from argus.llm.batch import enqueue_for_llm

logger = logging.getLogger(__name__)


def load_sector_map(path: str = "config/sectors.yaml") -> dict:
    if os.path.exists(path):
        return yaml.safe_load(open(path, encoding="utf-8")) or {}
    return {}


def _published(entry):
    pp = getattr(entry, "published_parsed", None)
    if not pp:
        return None
    return datetime.fromtimestamp(calendar.timegm(pp), tz=timezone.utc)


def _add_claim(session, source, item, asset_symbol, direction, confidence, method,
               parser_version, parsed_claim, raw_text, entry_price, now):
    claim = Claim(source_id=source.id, item_id=item.id, asset=asset_symbol,
                  direction=direction, confidence=confidence, method=method,
                  entry_price=entry_price, entry_ts=now, status="open")
    session.add(claim); session.flush()
    session.add(ClaimAudit(claim_id=claim.id, raw_text=raw_text,
                           parsed_claim=parsed_claim, parser_version=parser_version))


def collect_rss(session, source) -> int:
    feed = feedparser.parse(source.handle)
    assets = list(session.scalars(select(Asset).where(Asset.scorable == True)))  # noqa: E712
    by_symbol = {a.symbol: a for a in assets}
    sector_map = load_sector_map()
    now = datetime.now(timezone.utc)
    created = 0
    stats = {"items": 0, "rule_claims": 0, "enqueued": 0}
    for entry in feed.entries:
        ext = getattr(entry, "id", None) or getattr(entry, "link", None)
        if ext is None:
            continue
        if session.scalar(select(Item).where(Item.source_id == source.id, Item.external_id == ext)):
            continue
        text = f"{getattr(entry, 'title', '')} {getattr(entry, 'summary', '')}"
        item = Item(source_id=source.id, external_id=ext, title=getattr(entry, "title", None),
                    text=text, url=getattr(entry, "link", None),
                    published_at=_published(entry), fetched_at=now)
        session.add(item); session.flush()
        created += 1; stats["items"] += 1

        rule_pcs = extract_claims(text, assets, sector_map=sector_map)
        if rule_pcs:
            for pc in rule_pcs:
                asset = by_symbol[pc.asset_symbol]
                try:
                    spot = get_spot_price(asset)
                except Exception as e:
                    logger.warning("skip rule claim %s: price failed: %s", pc.asset_symbol, e)
                    continue
                _add_claim(session, source, item, asset.symbol, pc.direction, pc.confidence,
                           "rule", PARSER_VERSION, asdict(pc), text, spot, now)
                stats["rule_claims"] += 1
        elif source.kind in ("reddit", "news"):
            if enqueue_for_llm(session, item, source.kind, text):
                stats["enqueued"] += 1

    session.commit()
    logger.info("collect_rss %s: %s", source.display_name, stats)
    return created
```
